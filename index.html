    return `<li class="rank-item flex items-center gap-3 rounded-xl px-3 py-2.5 border border-white/5 ${me ? 'rank-me' : ''}">

      <span class="orbitron w-7 text-center text-sm font-bold ${rankStyle}">${medals[i] || i + 1}</span>

      <span class="flex-1 truncate text-sm font-medium ${me ? 'text-cyan-200' : ''}">${escapeHtml(r.name)}${me ? ' <span class="text-[10px] text-cyan-300">ME</span>' : ''}${r.win ? ' <span title="클리어">👑</span>' : ''}</span>

      <span class="orbitron text-sm font-bold text-white">${Number(r.score).toLocaleString()}</span>

    </li>`;

  }).join('');

  // 내 최고 기록 갱신 (Top10 내 있을 경우)

  if (player) {

    const mine = list.filter(r => r.name === player).map(r => r.score);

    if (mine.length) myBest = Math.max(myBest, ...mine);

  }

  $('myBest').textContent = myBest.toLocaleString();

}

function escapeHtml(s) { return String(s).replace(/[&<>"']/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c])); }


/* ══════════════════════════════════════════════════════════════

   게임 엔진

   ══════════════════════════════════════════════════════════════ */

const COLS = 10, ROWS = 6, BRICK_GAP = 6, BRICK_TOP = 60, BRICK_H = 22, SIDE_PAD = 24;

const BRICK_W = (W - SIDE_PAD * 2 - BRICK_GAP * (COLS - 1)) / COLS;

const ROW_COLORS = [

  ['#fb7185', '#f43f5e'], ['#f472b6', '#ec4899'], ['#a78bfa', '#8b5cf6'],

  ['#60a5fa', '#3b82f6'], ['#22d3ee', '#06b6d4'], ['#34d399', '#10b981']

];

const BASE_SPEED = 360, MAX_SPEED = 620, PADDLE_SPEED = 560;


const Game = {

  mode: 'idle',      // idle | playing | paused | over

  score: 0, lives: 3, combo: 0, maxCombo: 0, broken: 0,

  bricks: [], particles: [], stars: [],

  paddle: { x: W / 2 - 60, y: H - 34, w: 120, h: 14 },

  ball: { x: W / 2, y: 0, r: 8, vx: 0, vy: 0, speed: BASE_SPEED, trail: [] },

  keys: { left: false, right: false },

  raf: null, last: 0, saved: false, shake: 0, flash: 0,


  reset() {

    this.score = 0; this.lives = 3; this.combo = 0; this.maxCombo = 0; this.broken = 0;

    this.saved = false; this.particles = []; this.shake = 0; this.flash = 0;

    this.buildBricks();

    if (!this.stars.length) for (let i = 0; i < 60; i++) this.stars.push({ x: Math.random() * W, y: Math.random() * H, r: Math.random() * 1.4 + .3, a: Math.random() });

    this.resetBall();

    this.mode = 'idle';

    this.updateHUD();

    if (!this.raf) { this.last = performance.now(); this.raf = requestAnimationFrame(t => this.loop(t)); }

  },

  stop() { if (this.raf) cancelAnimationFrame(this.raf); this.raf = null; this.mode = 'idle'; },

  buildBricks() {

    this.bricks = [];

    for (let r = 0; r < ROWS; r++) for (let c = 0; c < COLS; c++) {

      this.bricks.push({

        x: SIDE_PAD + c * (BRICK_W + BRICK_GAP), y: BRICK_TOP + r * (BRICK_H + BRICK_GAP),

        w: BRICK_W, h: BRICK_H, row: r, alive: true, points: (ROWS - r) * 10, hp: r === 0 ? 2 : 1

      });

    }

  },

  resetBall() {

    const b = this.ball;

    b.x = this.paddle.x + this.paddle.w / 2; b.y = this.paddle.y - b.r - 1;

    b.vx = 0; b.vy = 0; b.trail = [];

    b.speed = Math.min(MAX_SPEED, BASE_SPEED + this.broken * 3);

  },

  launch() {

    if (this.mode !== 'idle') return;

    const ang = (-Math.PI / 2) + (Math.random() * .6 - .3);

    this.ball.vx = Math.cos(ang) * this.ball.speed;

    this.ball.vy = Math.sin(ang) * this.ball.speed;

    this.mode = 'playing';

    this.hideOverlay();

  },

  togglePause() {

    if (this.mode === 'playing') { this.mode = 'paused'; this.showOverlay('paused'); }

    else if (this.mode === 'paused') { this.mode = 'playing'; this.hideOverlay(); }

  },


  /* ── 오버레이 ── */

  showOverlay(type, data = {}) {

    const set = (id, v) => $(id).textContent = v;

    const stats = $('ovStats'), sec = $('ovSecondary'), pri = $('ovPrimary');

    stats.classList.add('hidden'); sec.classList.add('hidden'); $('ovSave').textContent = '';

    if (type === 'ready') {

      set('ovEmoji', '🎮'); set('ovTitle', 'READY?'); set('ovDesc', '스페이스 키 또는 ▲ 키를 눌러 공을 발사하세요.'); set('ovPrimary', '시작하기');

      pri.onclick = () => this.launch();

    } else if (type === 'paused') {

      set('ovEmoji', '⏸️'); set('ovTitle', 'PAUSED'); set('ovDesc', 'P 키를 눌러 계속 진행하세요.'); set('ovPrimary', '계속하기');

      pri.onclick = () => this.togglePause();

    } else if (type === 'life') {

      set('ovEmoji', '💔'); set('ovTitle', `LIVES: ${this.lives}`); set('ovDesc', '아직 기회가 있어요! 스페이스 키로 다시 발사하세요.'); set('ovPrimary', '다시 발사');

      pri.onclick = () => this.launch();

    } else if (type === 'win' || type === 'lose') {

      set('ovEmoji', type === 'win' ? '🏆' : '💥');

      set('ovTitle', type === 'win' ? 'STAGE CLEAR!' : 'GAME OVER');

      set('ovDesc', type === 'win' ? `모든 벽돌을 깼습니다! 최대 콤보 x${this.maxCombo}` : `최대 콤보 x${this.maxCombo} · 다음엔 더 잘할 수 있어요!`);

      stats.classList.remove('hidden');

      set('ovScore', this.score.toLocaleString()); set('ovBonus', '+' + (data.bonus || 0).toLocaleString());

      set('ovPrimary', '다시 하기'); pri.onclick = () => { this.reset(); this.showOverlay('ready'); };

      sec.classList.remove('hidden'); sec.textContent = '순위 보기';

      sec.onclick = () => $('leaderboard').scrollIntoView({ behavior: 'smooth', block: 'center' });

    }

    overlay.classList.remove('hidden-ov');

  },

  hideOverlay() { overlay.classList.add('hidden-ov'); },


  updateHUD() {

    $('hudScore').textContent = this.score.toLocaleString();

    $('hudLives').textContent = '❤️'.repeat(Math.max(0, this.lives)) + '🖤'.repeat(Math.max(0, 3 - this.lives));

    $('hudBricks').textContent = this.bricks.filter(b => b.alive).length;

    $('hudCombo').textContent = 'x' + Math.max(1, this.combo);

  },


  /* ── 종료 처리 ── */

  async endGame(win) {

    this.mode = 'over';

    const bonus = win ? this.lives * 150 + 300 : 0;

    this.score += bonus;

    this.updateHUD();

    this.showOverlay(win ? 'win' : 'lose', { bonus });

    if (win) this.burst(W / 2, H / 2, 80, ['#f472b6', '#a78bfa', '#22d3ee', '#fcd34d', '#34d399']);

    if (this.saved || !player) return;

    this.saved = true;

    const saveEl = $('ovSave');

    saveEl.textContent = '점수 저장 중...';

    try {

      await Store.saveScore({ name: player, score: this.score, win, maxCombo: this.maxCombo });

      myBest = Math.max(myBest, this.score); $('myBest').textContent = myBest.toLocaleString();

      saveEl.textContent = Store.mode === 'firebase' ? '✅ Firebase에 점수가 저장되었습니다.' : '✅ 점수가 저장되었습니다. (로컬 모드)';

    } catch (e) {

      console.error(e); saveEl.textContent = '⚠️ 점수 저장에 실패했습니다. DB 규칙을 확인하세요.';

    }

  },


  /* ── 파티클 ── */

  burst(x, y, n, colors) {

    for (let i = 0; i < n; i++) {

      const a = Math.random() * Math.PI * 2, s = 60 + Math.random() * 220;

      this.particles.push({ x, y, vx: Math.cos(a) * s, vy: Math.sin(a) * s, life: .5 + Math.random() * .5, max: 1, r: 1.5 + Math.random() * 3, c: colors[Math.floor(Math.random() * colors.length)] });

    }

  },


  /* ── 업데이트 ── */

  update(dt) {

    const p = this.paddle, b = this.ball;

    // 패들 이동

    if (this.mode === 'playing' || this.mode === 'idle') {

      if (this.keys.left) p.x -= PADDLE_SPEED * dt;

      if (this.keys.right) p.x += PADDLE_SPEED * dt;

      p.x = Math.max(8, Math.min(W - p.w - 8, p.x));

    }

    if (this.mode === 'idle') { b.x = p.x + p.w / 2; b.y = p.y - b.r - 1; }

    // 파티클

    for (const q of this.particles) { q.x += q.vx * dt; q.y += q.vy * dt; q.vy += 300 * dt; q.life -= dt; }

    this.particles = this.particles.filter(q => q.life > 0);

    this.shake = Math.max(0, this.shake - dt * 30);

    this.flash = Math.max(0, this.flash - dt * 3);

    if (this.mode !== 'playing') return;


    // 공 이동 (서브스텝으로 터널링 방지)

    const steps = Math.ceil(Math.hypot(b.vx, b.vy) * dt / 6);

    const sdt = dt / steps;

    for (let s = 0; s < steps; s++) {

      b.x += b.vx * sdt; b.y += b.vy * sdt;

      // 벽

      if (b.x - b.r < 0) { b.x = b.r; b.vx = Math.abs(b.vx); }

      if (b.x + b.r > W) { b.x = W - b.r; b.vx = -Math.abs(b.vx); }

      if (b.y - b.r < 0) { b.y = b.r; b.vy = Math.abs(b.vy); }

      // 패들

      if (b.vy > 0 && b.y + b.r >= p.y && b.y + b.r <= p.y + p.h + 10 && b.x >= p.x - b.r && b.x <= p.x + p.w + b.r) {

        const hit = ((b.x - (p.x + p.w / 2)) / (p.w / 2));

        const ang = hit * (Math.PI / 3);   // 최대 60도

        b.vx = Math.sin(ang) * b.speed; b.vy = -Math.cos(ang) * b.speed;

        b.y = p.y - b.r;

        this.combo = 0; this.updateHUD();

      }

      // 벽돌

      for (const br of this.bricks) {

        if (!br.alive) continue;

        const cx = Math.max(br.x, Math.min(b.x, br.x + br.w));

        const cy = Math.max(br.y, Math.min(b.y, br.y + br.h));

        const dx = b.x - cx, dy = b.y - cy;

        if (dx * dx + dy * dy <= b.r * b.r) {

          // 충돌 방향 판정

          const overlapX = b.r - Math.abs(dx), overlapY = b.r - Math.abs(dy);

          if (Math.abs(dx) > Math.abs(dy) && dx !== 0) { b.vx = dx > 0 ? Math.abs(b.vx) : -Math.abs(b.vx); b.x += dx > 0 ? overlapX : -overlapX; }

          else { b.vy = dy > 0 ? Math.abs(b.vy) : -Math.abs(b.vy); b.y += dy > 0 ? overlapY : -overlapY; }

          br.hp--;

          if (br.hp <= 0) {

            br.alive = false; this.broken++;

            this.combo++; this.maxCombo = Math.max(this.maxCombo, this.combo);

            this.score += br.points * Math.max(1, this.combo);

            this.burst(br.x + br.w / 2, br.y + br.h / 2, 14, ROW_COLORS[br.row]);

            b.speed = Math.min(MAX_SPEED, BASE_SPEED + this.broken * 3);

            const m = Math.hypot(b.vx, b.vy); b.vx = b.vx / m * b.speed; b.vy = b.vy / m * b.speed;

          } else {

            this.score += 5; this.burst(b.x, b.y, 5, ['#ffffff']);

          }

          this.shake = 4; this.updateHUD();

          if (!this.bricks.some(x => x.alive)) { this.endGame(true); return; }

          break;

        }

      }

      // 낙하

      if (b.y - b.r > H) {

        this.lives--; this.combo = 0; this.flash = 1; this.shake = 8; this.updateHUD();

        if (this.lives <= 0) { this.resetBall(); this.endGame(false); return; }

        this.resetBall(); this.mode = 'idle'; this.showOverlay('life'); return;

      }

    }

    // 수직 속도가 너무 작아지는 것 방지

    if (Math.abs(b.vy) < b.speed * .2) { const sign = b.vy < 0 ? -1 : 1; b.vy = sign * b.speed * .2; const m = Math.hypot(b.vx, b.vy); b.vx = b.vx / m * b.speed; b.vy = b.vy / m * b.speed; }

    b.trail.push({ x: b.x, y: b.y }); if (b.trail.length > 10) b.trail.shift();

  },


  /* ── 렌더 ── */

  draw() {

    ctx.save();

    ctx.clearRect(0, 0, W, H);

    if (this.shake > 0) ctx.translate((Math.random() - .5) * this.shake, (Math.random() - .5) * this.shake);

    // 별

    for (const s of this.stars) { s.a += .01; ctx.globalAlpha = .3 + Math.abs(Math.sin(s.a)) * .5; ctx.fillStyle = '#c7d2fe'; ctx.beginPath(); ctx.arc(s.x, s.y, s.r, 0, 7); ctx.fill(); }

    ctx.globalAlpha = 1;

    // 상단 라인

    const g = ctx.createLinearGradient(0, 0, W, 0); g.addColorStop(0, 'rgba(244,114,182,0)'); g.addColorStop(.5, 'rgba(167,139,250,.8)'); g.addColorStop(1, 'rgba(34,211,238,0)');

    ctx.fillStyle = g; ctx.fillRect(0, 0, W, 2);

    // 벽돌

    for (const br of this.bricks) {

      if (!br.alive) continue;

      const [c1, c2] = ROW_COLORS[br.row];

      const bg = ctx.createLinearGradient(br.x, br.y, br.x, br.y + br.h); bg.addColorStop(0, c1); bg.addColorStop(1, c2);

      ctx.shadowColor = c1; ctx.shadowBlur = 12;

      ctx.fillStyle = bg; roundRect(br.x, br.y, br.w, br.h, 6); ctx.fill();

      ctx.shadowBlur = 0;

      ctx.fillStyle = 'rgba(255,255,255,.28)'; roundRect(br.x + 3, br.y + 3, br.w - 6, 5, 3); ctx.fill();

      if (br.hp > 1) { ctx.strokeStyle = 'rgba(255,255,255,.7)'; ctx.lineWidth = 1.5; roundRect(br.x + 1, br.y + 1, br.w - 2, br.h - 2, 5); ctx.stroke(); }

    }

    // 공 트레일

    const b = this.ball;

    b.trail.forEach((t, i) => { ctx.globalAlpha = (i / b.trail.length) * .35; ctx.fillStyle = '#a5f3fc'; ctx.beginPath(); ctx.arc(t.x, t.y, b.r * (i / b.trail.length), 0, 7); ctx.fill(); });

    ctx.globalAlpha = 1;

    // 공

    ctx.shadowColor = '#22d3ee'; ctx.shadowBlur = 20;

    const bgd = ctx.createRadialGradient(b.x - 3, b.y - 3, 1, b.x, b.y, b.r); bgd.addColorStop(0, '#ffffff'); bgd.addColorStop(1, '#22d3ee');

    ctx.fillStyle = bgd; ctx.beginPath(); ctx.arc(b.x, b.y, b.r, 0, 7); ctx.fill();

    ctx.shadowBlur = 0;

    // 패들

    const p = this.paddle;

    const pg = ctx.createLinearGradient(p.x, 0, p.x + p.w, 0); pg.addColorStop(0, '#f472b6'); pg.addColorStop(.5, '#a78bfa'); pg.addColorStop(1, '#22d3ee');

    ctx.shadowColor = '#a78bfa'; ctx.shadowBlur = 18;

    ctx.fillStyle = pg; roundRect(p.x, p.y, p.w, p.h, 8); ctx.fill();

    ctx.shadowBlur = 0;

    ctx.fillStyle = 'rgba(255,255,255,.35)'; roundRect(p.x + 6, p.y + 3, p.w - 12, 4, 2); ctx.fill();

    // 파티클

    for (const q of this.particles) { ctx.globalAlpha = Math.max(0, q.life / q.max); ctx.fillStyle = q.c; ctx.beginPath(); ctx.arc(q.x, q.y, q.r, 0, 7); ctx.fill(); }

    ctx.globalAlpha = 1;

    // 피격 플래시

    if (this.flash > 0) { ctx.fillStyle = `rgba(244,63,94,${this.flash * .25})`; ctx.fillRect(-10, -10, W + 20, H + 20); }

    ctx.restore();

  },

  loop(t) {

    const dt = Math.min(.033, (t - this.last) / 1000); this.last = t;

    this.update(dt); this.draw();

    this.raf = requestAnimationFrame(tt => this.loop(tt));

  }

};

function roundRect(x, y, w, h, r) {

  ctx.beginPath();

  ctx.moveTo(x + r, y); ctx.lineTo(x + w - r, y); ctx.quadraticCurveTo(x + w, y, x + w, y + r);

  ctx.lineTo(x + w, y + h - r); ctx.quadraticCurveTo(x + w, y + h, x + w - r, y + h);

  ctx.lineTo(x + r, y + h); ctx.quadraticCurveTo(x, y + h, x, y + h - r);

  ctx.lineTo(x, y + r); ctx.quadraticCurveTo(x, y, x + r, y); ctx.closePath();

}

window.addEventListener('keydown', e => {

  if (!player) return;

  if (e.target.tagName === 'INPUT') return;

  const k = e.key;

  if (['ArrowLeft', 'ArrowRight', 'ArrowUp', ' '].includes(k)) e.preventDefault();

  if (k === 'ArrowLeft' || k === 'a' || k === 'A') Game.keys.left = true;

  if (k === 'ArrowRight' || k === 'd' || k === 'D') Game.keys.right = true;

  if (k === ' ' || k === 'ArrowUp') { if (Game.mode === 'idle') Game.launch(); else if (Game.mode === 'over') { Game.reset(); Game.showOverlay('ready'); } }

  if (k === 'p' || k === 'P' || k === 'Escape') Game.togglePause();

  if (k === 'r' || k === 'R') { Game.reset(); Game.showOverlay('ready'); }

});

window.addEventListener('keyup', e => {

  const k = e.key;

  if (k === 'ArrowLeft' || k === 'a' || k === 'A') Game.keys.left = false;

  if (k === 'ArrowRight' || k === 'd' || k === 'D') Game.keys.right = false;

});
